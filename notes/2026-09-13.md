## 常用命令速查(今天实际用过的)

```powershell
docker build -t todo-api .                          # 构建镜像,观察 => CACHED 分层缓存
docker run -d --name todo-api -p 8000:8000 `
  -e TODO_DATABASE_URL=postgresql+psycopg://postgres@host.docker.internal:5432/todo `
  -e TODO_REDIS_URL=redis://host.docker.internal:6379/0 todo-api
docker volume create demo-vol                       # 卷实验:两个 --rm 容器接力读写
docker network create todo-net
docker compose up -d db cache                       # 先起基础设施,等 healthy
docker compose run --rm api alembic upgrade head    # 一次性容器跑迁移(部署标准动作)
docker compose up -d                                # 全栈一条命令
docker compose exec db psql -U postgres -d todo -c "\dt"
docker compose exec cache redis-cli GET todos:list
docker compose exec api python -c "import urllib.request; print(urllib.request.urlopen('http://api:8000/todos').status)"  # 服务名=域名
docker compose down / down -v                       # 删容器 / 连数据卷一起删
docker compose up -d --build                        # entrypoint 变更后重建
docker compose logs api | grep -i alembic           # 验证 entrypoint 自动跑了迁移
```
